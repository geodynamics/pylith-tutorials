\begin{frame}{Bayes' Rule}\Huge
\begin{align*}
  P(H | D) = \frac{P(D | H) P(H)}{P(D)}
\end{align*}
\end{frame}
%
\begin{frame}{Bayes' Rule}\Huge
\begin{align*}
  P(H | D) = \frac{P(D | H) P(H)}{\sum_i P(D | H_i) P(H_i)}
\end{align*}
\end{frame}
%
